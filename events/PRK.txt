			is_decaying = yes

		set_state_religion = rel:atheist
		add_modifier = {
			name = antireligious_policies
			years = 10
		}
		custom_tooltip = {
			text = state_atheism_conversion_tt
			every_scope_state = {
				limit = {
					is_incorporated = yes
				}
				convert_population = {
					target = rel:atheist
					value = 0.15
				}
			}
		}				
		change_relations = { country = c:RUS value = 100 }}
		create_character = { #金日成
    		dna = ecchi_dsk_dna_jinricheng		
			first_name = "Richeng"
			last_name = "Jin"
			historical = yes
			ruler = yes # both ruler and general
			birth_date = 1912.4.15
			religion = rel:atheist #无神论
			ig_leader = yes
			interest_group = ig_armed_forces #军队
			ideology = ideology_communist #共产主义者
			set_character_immortal = yes			
			traits = {
				kindness #受人崇拜
				expensive_tastes #昂贵嗜好
			}	
		}	
		create_character = { #金正日
    		dna = ecchi_dsk_dna_jinricheng		
			first_name = "Zhengri"
			last_name = "Jin"
			historical = yes
			heir = yes	
			birth_date = 1942.2.16
			religion = rel:atheist #无神论
			interest_group = ig_armed_forces #军队
			ideology = ideology_communist #共产主义者
			set_character_immortal = yes			
			traits = {
				kindness #受人崇拜

       		trigger_event = { id = PRK_event.3 } 		
	}
}
