guoyou.1 = {
	type = country_event
	hidden = yes
	placement = ROOT

	trigger = {
		OR = {
			has_law = law_type:law_laissez_faire
			has_law = law_type:law_command_economy
		}
		OR = {
			has_modifier = adjust_reinvestment
			has_modifier = tax_modifier_low
			has_modifier = tax_modifier_medium
			has_modifier = tax_modifier_high
			has_modifier = tax_modifier_very_high
		}
	}
	
	option = {
		name = "Destroy them all"
		remove_modifier = adjust_reinvestment
		remove_modifier = tax_modifier_low
		remove_modifier = tax_modifier_medium
		remove_modifier = tax_modifier_high
		remove_modifier = tax_modifier_very_high
	}
}
