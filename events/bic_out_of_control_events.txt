# 时间耗尽
bic_out_of_control.3 = {
	type = country_event
	placement = c:MUG.capital

	title = bic_out_of_control.3.t
	desc = bic_out_of_control.3.d
	flavor = bic_out_of_control.3.f

	duration = 3

	event_image = {
		video = "asia_sepoy_mutiny"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	immediate = {
		remove_modifier = bic_out_of_control_modifier
		c:GBR = {
			remove_modifier = gbr_colony_india_modifier
		}		
		c:BIC = {
			every_scope_state = {
				limit = {
					is_incorporated = yes
				}
				set_variable = bic_incorporated_state
			}
		}
		# 印度分离，以孟买作为首都
		if = {
			limit = {
				NOT = { exists = c:HND }
				exists = s:STATE_BOMBAY.region_state:BIC
			}
			create_country = {
				tag = HND
				origin = c:BIC
				state = s:STATE_BOMBAY.region_state:BIC
			}
		}
		else_if = {
			limit = {
				exists = c:HND
				exists = s:STATE_BOMBAY.region_state:BIC
			}
			s:STATE_BOMBAY.region_state:BIC = { set_state_owner = c:HND }
		}
		# 巴基斯坦分离，以东旁遮普为首都
		if = {
			limit = {
				NOT = { exists = c:PAK }
				exists = s:STATE_PUNJAB.region_state:BIC
			}
			create_country = {
				tag = PAK
				origin = c:BIC
				state = s:STATE_PUNJAB.region_state:BIC
			}
		}
		else_if = {
			limit = {
				exists = c:PAK
				exists = s:STATE_PUNJAB.region_state:BIC
			}
			s:STATE_PUNJAB.region_state:BIC = { set_state_owner = c:HND }
		}
		# 分配地块给巴基斯坦
		if = {
			limit = {
				exists = c:PAK
				exists = s:STATE_BALUCHISTAN.region_state:BIC
			}
			s:STATE_BALUCHISTAN.region_state:BIC = { set_state_owner = c:PAK }
		}
		if = {
			limit = {
				exists = c:PAK
				exists = s:STATE_PASHTUNISTAN.region_state:BIC
			}
			s:STATE_PASHTUNISTAN.region_state:BIC = { set_state_owner = c:PAK }
		}
		if = {
			limit = {
				exists = c:PAK
				exists = s:STATE_KASHMIR.region_state:BIC
			}
			s:STATE_KASHMIR.region_state:BIC = { set_state_owner = c:PAK }
		}
		if = {
			limit = {
				exists = c:PAK
				exists = s:STATE_SINDH.region_state:BIC
			}
			s:STATE_SINDH.region_state:BIC = { set_state_owner = c:PAK }
		}
		c:BIC = {
			set_capital = STATE_DELHI
		}
		# 分配地块给印度
		every_scope_state = {
			limit = {
				is_capital = no
				owner = c:BIC
			}
			set_state_owner = c:HND
		}
		if = {
			limit = {
				NOT = {
					any_diplomatic_play = {
						any_scope_play_involved = { this = c:HND }
						is_diplomatic_play_type = dp_secession
					}
				}
			}
			create_diplomatic_play = {
				name = sepoy_mutiny
				type = dp_annex_war
				escalation = 10
				target_state = s:STATE_BOMBAY.region_state:HND
			}
			random_diplomatic_play = {
				limit = {
					any_scope_play_involved = {
						c:BIC ?= this
					}
				}
				if = {
					limit = {
						exists = s:STATE_DELHI.region_state:BIC 
					}
					add_war_goal = {
						holder = c:HND
						type = conquer_state
						target_state = s:STATE_DELHI.region_state:BIC
					}
				}
			}
		}
		if = {
			limit = {
				NOT = {
					any_diplomatic_play = {
						any_scope_play_involved = { this = c:PAK }
						is_diplomatic_play_type = dp_secession
					}
				}
			}
			create_diplomatic_play = {
				name = sepoy_mutiny
				type = dp_annex_war
				escalation = 10
				target_state = s:STATE_PUNJAB.region_state:PAK
			}
			random_diplomatic_play = {
				limit = {
					any_scope_play_involved = {
						c:BIC ?= this
					}
				}
				if = {
					limit = {
						exists = s:STATE_DELHI.region_state:BIC 
					}
					add_war_goal = {
						holder = c:PAK
						type = conquer_state
						target_state = s:STATE_DELHI.region_state:BIC
					}
				}
			}
		}
		add_modifier = {
			name = weakened_by_mutiny
			months = normal_modifier_time
		}
		c:HND = {
            activate_law = law_type:law_parliamentary_republic #改变法律为议会共和制
			activate_law = law_type:law_interventionism #改变法律为经济干预	
			activate_law = law_type:law_protectionism #改变法律为贸易保护
        	trigger_event = { id = bic.1 } #领导权问题				
		create_character = {
			ig_leader = yes #利益集团领袖
			religion = rel:hindu #印度教
			interest_group = ig_devout #虔信者
			traits = {
			}
		}

		create_character = {
			ig_leader = yes #利益集团领袖
			religion = rel:hindu #印度教
			interest_group = ig_armed_forces #军队
		}

		create_character = {
			ig_leader = yes
			religion = rel:hindu #印度教
			interest_group = ig_landowners #地主
		}

		create_character = {
			ig_leader = yes #利益集团领袖
			religion = rel:hindu #印度教			
			interest_group = ig_trade_unions #利益集团工会
		}	
		}
		c:PAK = {
			activate_law = law_type:law_presidential_republic #改变法律为总统共和制
			activate_law = law_type:law_interventionism #改变法律为经济干预	
			activate_law = law_type:law_protectionism #改变法律为贸易保护				
		create_character = { #巴基斯坦总统穆罕默德·阿里·真纳
			historical = yes
			religion = rel:shiite #什叶派
			culture = cu:gujarati
			first_name = "Muhammad_Ali"
			last_name = "Jinnah"
			ruler = yes
			ig_leader = yes
			interest_group = ig_intelligentsia
    		dna = dna_muhammad_ali_jinnah
			ideology = ideology_social_democrat
			birth_date = 1876.12.25
			traits = {
				erudite #博学
				inspirational_orator #鼓舞人心的演说家
				firebrand #煽动大师
				pious #虔诚	
				sickly #多病
			}
		}

		create_character = {
			ig_leader = yes #利益集团领袖
			culture = cu:panjabi			
			religion = rel:sunni #逊尼派
			interest_group = ig_devout #虔信者
			traits = {
			}
		}

		create_character = {
			ig_leader = yes #利益集团领袖
			culture = cu:panjabi			
			religion = rel:sunni #逊尼派
			interest_group = ig_armed_forces #军队
		}

		create_character = {
			ig_leader = yes #利益集团领袖
			culture = cu:panjabi			
			religion = rel:sunni #逊尼派
			interest_group = ig_landowners #地主
		}

		create_character = {
			ig_leader = yes #利益集团领袖
			culture = cu:panjabi			
			religion = rel:sunni #逊尼派	
			interest_group = ig_trade_unions #利益集团工会
		}	

		create_character = {
			ig_leader = yes #利益集团领袖
			culture = cu:panjabi			
			religion = rel:sunni #逊尼派
			interest_group = ig_rural_folk #农民
			traits = {
			}
		}		

		create_character = {
			ig_leader = yes #利益集团领袖
			culture = cu:panjabi			
			religion = rel:sunni #逊尼派
			interest_group = ig_industrialists #实业家
			ideology = ideology_utilitarian_leader #功利主义者
			traits = {
			}
		}	

		create_character = {
			ig_leader = yes #利益集团领袖
			culture = cu:panjabi			
			religion = rel:sunni #逊尼派
			interest_group = ig_petty_bourgeoisie #小市民
		}			
		}
	}

	option = {
		name = bic_out_of_control.3.a
		default_option = yes
	}
}
