west_germany_event.1 = { 六国伦敦会议
	type = country_event
	orphan = yes
	
	title = BRD.1.t
	desc = BRD.1.d
	flavor = BRD.1.f

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/BRD.dds"

	duration = 3
	
trigger = { 
	trigger_if ={
            limit ={ c:GBR  = { is_player= yes }
	}
			c:GBR ?= THIS
 }
    trigger_else_if ={
            limit ={ c:FRA = { is_player= yes }
	}
			c:FRA ?= THIS
 }
    trigger_else ={
            c:USA ?= THIS
 } 	
    NOT = { has_global_variable = brd_setup }
}

immediate = { 
	c:USA = {
		save_scope_as = usa_c_scope
	}
	c:GBR = {
		save_scope_as = uk_c_scope
	}
	c:FRA = {
		save_scope_as = fra_c_scope
	}
	c:PRC = {
		save_scope_as = prc_c_scope
	}
	set_global_variable = brd_setup
}
	option = {
		name = BRD.1.a
            c:USA ={
            change_relations = {
				country = c:RUS
				value = 20
			}
	}
	c:FRA ={
            change_relations = {
				country = c:RUS
				value = 20
			}
	}
	c:GBR ={
            change_relations = {
				country = c:RUS
				value = 20
			}
	}
		ai_chance = {
         base = 50
		}
	}

	option = {
		name = BRD.1.b
		add_modifier = {
			name = the_country_of_our_ancestors_modifier
			years = 1
		}
            c:USA ={
            change_relations = {
				country = c:RUS
				value = -20
			}
	    }
	c:RUS ={ trigger_event = { id = west_germany_event.2 popup = yes }}
	ai_chance = {
         base = 50
		}
	}
}

	option = {
		name = BRD.4.a
		add_modifier = {
			name = the_country_of_our_ancestors_modifier
			years = 1
		}
			if ={
            limit ={
                OR ={
                    AND ={
                        exists =C:RHN
                        this=c:RHN
                        c:RHN ?= {is_player= yes}
			}
                    AND ={
                        exists =C:YSD

                        c:YSD ?= {is_player= yes}
			}
		}
	}
			play_as = c:BAV
	}
            c:BAV ={
            remove_primary_culture = cu:south_german
            add_primary_culture = cu:north_german

		          annex = c:RHN

				  make_independent = yes
	set_capital = STATE_NORTH_RHINE

			}
		create_character = {
			is_general = yes
			ig_leader = yes
			interest_group = ig_armed_forces #军队
		}
