salazar's_funeral = { #萨拉查的葬礼

	group = je_group_internal_affairs

	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	is_shown_when_inactive = { #显示条件
			c:POR ?= ROOT
		    exists = c:POR								
	}   
	possible = { #激活条件
        exists = c:POR
        THIS = c:POR
		game_date >= 1970.6.27					
	}
	complete = { #完成条件
		game_date >= 1970.7.27
	}
    on_complete = {
        trigger_event = { id = por.1 } #游行队伍（萨拉查去世）        
        trigger_event = { id = por.2 } #再见了，考迪罗
    }

    weight = 1000
    should_be_pinned_by_default = yes
}
