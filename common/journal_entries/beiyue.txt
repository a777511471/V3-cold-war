	group = je_group_foreign_affairs

	icon = "gfx/interface/icons/event_icons/active_peace_deal.dds"

	is_shown_when_inactive = { #激活条件
			is_ai = no	
		NOT = { has_law = law_type:law_council_republic }
			c:USA ?= ROOT	
		    exists = c:USA								
	}
	
	possible = {
			is_ai = no	
		NOT = { has_law = law_type:law_council_republic }		
			c:USA ?= ROOT	
		    exists = c:USA		
	}

	complete = { #完成条件
		NOT = {
        	has_law = law_type:law_council_republic
		}			
        c:GBR = { 
            relations:root >= 50 
        }	
        c:FRA = { 
            relations:root >= 50 
        }	
		custom_tooltip = {
			text = custom_loc_shichangyitihua_goal
				scope:journal_entry = {
				is_goal_complete = yes
			}
		}        
		in_default = no	
		NOT = {
        	has_law = law_type:law_non_militarized_country
			has_law = law_type:law_peasant_levies
		}				
		army_size > 500	
	}
    on_complete = {
		trigger_event = {
			id = guojialianmeng.2
			popup = yes
	    }
	}

	current_value = {
		value = root.gdp
	}

	goal_add_value = {
		value = root.gdp
		multiply = 0.25
	}

	fail = {
		has_law = law_type:law_council_republic 
	}

    weight = 1000
	progressbar = yes	
    should_be_pinned_by_default = yes
}
