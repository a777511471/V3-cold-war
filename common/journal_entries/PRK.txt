je_prk = { #统一朝鲜
	icon = "gfx/interface/icons/event_icons/统一朝鲜.dds"

	group = je_group_foreign_affairs

	modifiers_while_active = {
		modifier_chinese_warlord_era
	}

	is_shown_when_inactive = {
		OR = {
			c:PRK ?= ROOT
			c:KOR ?= ROOT
		}
	}
	
	possible = {
		OR = {
			c:PRK ?= ROOT
			c:KOR ?= ROOT
		}
	}

	complete = {
		trigger_if = {
			limit = {
				global_var:chinese_central_government ?= this
			}
			custom_tooltip = {
				text = no_independent_prk_countries_tt
				any_country = {
					filter = {
							country_has_primary_culture = cu:korean
							capital.region = sr:region_manchuria
					}
					OR = {
						THIS = ROOT
						is_subject_of = ROOT
					}
					percent = 1
				}
			}
		}
		trigger_else = {
			trigger_if = {
				limit = {
					global_var:chinese_central_government ?= {
						is_country_alive = yes
					}
				}
				custom_tooltip = {
					text = central_government_defeated_tt
					OR = {
						NOT = {
							exists = global_var:chinese_central_government
						}
						global_var:chinese_central_government ?= {
							is_country_alive = no
						}
					}
				}
			}
			custom_tooltip = {
				text = no_independent_prk_countries_tt
				any_country = {
					filter = {
							country_has_primary_culture = cu:korean
							capital.region = sr:region_manchuria
					}
					OR = {
						THIS = ROOT
						is_subject_of = ROOT
					}
					percent = 1
				}
			}
		}
		OR = {
			owns_entire_state_region = STATE_PYONGYANG
			owns_entire_state_region = STATE_SEOUL
		}
	}

	on_complete = {
		if = {
			limit = {
				has_modifier = loyalty_modifier
			}
			remove_modifier = loyalty_modifier
		}
		if = {
			limit = {
				has_modifier = the_general's_kindness_is_not_over_yet_modifier
			}
			remove_modifier = the_general's_kindness_is_not_over_yet_modifier
		}		
		remove_global_variable = chinese_central_government
		trigger_event = { id = chi.11 popup = yes }
	}
		
	weight = 1000
    should_be_pinned_by_default = yes	
}

peaceful_reunification_of_korea = { #朝鲜和平统一方案

	group = je_group_foreign_affairs

	icon = "gfx/interface/icons/event_icons/统一门.dds"

	is_shown_when_inactive = {
		OR = {
			c:PRK ?= ROOT
			c:KOR ?= ROOT
		}
		    exists = c:PRK		
		    exists = c:KOR			
	}
	
	possible = {
		OR = {
			c:PRK ?= ROOT
			c:KOR ?= ROOT
		}
		    exists = c:PRK		
		    exists = c:KOR			
	}

	complete = { #完成条件
		NOT = {
        	has_law = law_type:law_autocracy
		}			
	OR = {
        c:PRK = { 
            relations:root >= 50 
        }
        c:KOR = { 
            relations:root >= 50 
        }
	}
        c:USA = { 
            relations:root >= 50 
        }	
        c:PRC = { 
            relations:root >= 50 
        }	
		custom_tooltip = {
			text = 100_goal
				scope:journal_entry = {
				is_goal_complete = yes
			}
		}
		in_default = no	
		NOT = {
        	has_law = law_type:law_non_militarized_country
			has_law = law_type:law_peasant_levies
		}				
		army_size > 30	
	}
    on_complete = {
		trigger_event = {
			id = PRK_event.5
			popup = yes
	    }					
    }

	fail = { #失败条件某方不存在
		custom_tooltip = {
			text = peaceful_reunification_of_korea_goal
			OR = {
		    	NOT = { exists = c:PRK }		 
		    	NOT = { exists = c:KOR }	
			}				
		}	
    }

    on_fail = { #如果失败
    }

	current_value = {
		value = root.gdp
	}

	goal_add_value = {
		value = root.gdp
		multiply = 1
	}

	weight = 10000
    progressbar = yes	
	should_be_pinned_by_default = yes
}
