	is_shown_when_inactive = { #显示条件
        THIS = c:GBR				
	}
	possible = { #激活条件
        THIS = c:GBR						
	}
	complete = { #完成条件
    	NOT = {
        	has_law = law_type:law_industry_banned #禁止工业		
        	has_law = law_type:law_laissez_faire #自由放任	
		}		
		any_scope_state = {
			state_region = s:STATE_HOME_COUNTIES
			is_incorporated = yes
            any_scope_building = {
                is_building_type = building_steel_mills #合金加工厂     
                level >= 10
            }   
		}	
		any_scope_state = {
			state_region = s:STATE_LANCASHIRE
			is_incorporated = yes
            any_scope_building = {
                is_building_type = building_steel_mills #合金加工厂     
                level >= 10
            }   
		}		
		any_scope_state = {
			state_region = s:STATE_WEST_COUNTRY
			is_incorporated = yes
            any_scope_building = {
                is_building_type = building_steel_mills #合金加工厂     
                level >= 10
            }   
		}					
		custom_tooltip = {
			text = custom_loc_five_year_plan_goal
				scope:journal_entry = {
				is_goal_complete = yes
			}
		}
		in_default = no		
	}

	on_complete = { #如果完成
		add_modifier = {
			name = restoration_of_the_steel_industry #恢复钢铁业
			years = 20
			is_decaying = yes						
		}
	}

	current_value = {
		value = root.gdp
	}

	goal_add_value = {
		value = root.gdp
		multiply = 0.25
	}

	progressbar = yes
}
