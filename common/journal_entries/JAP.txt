

responding_to_the_energy_crisis = {  #应对能源危机
	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	group = je_group_internal_affairs

    is_shown_when_inactive = { #显示条件
		c:JAP ?= ROOT	
    }
	possible = { #激活条件
		c:JAP ?= ROOT		
	}

	complete = {
		any_scope_state = {
			is_incorporated = yes
			any_scope_building = {
				is_building_type = building_coal_mine #煤矿
				level >= 21
			}
        }			
		any_scope_state = {
			is_incorporated = yes
			any_scope_building = {
				is_building_type = building_iron_mine #金属矿
				level >= 21
			}
		}		
		any_scope_state = {
			is_incorporated = yes
			any_scope_building = {
				OR = {
					is_building_type = building_sulfur_mine #非金属矿
					is_building_type = building_sand_mine #砂石土矿 
				}
				level >= 21
			}
		}	
		any_scope_state = {
			is_incorporated = yes
			any_scope_building = {
				is_building_type = building_oil_rig #油井
				level >= 15
			}
		}							
		custom_tooltip = {
			text = custom_loc_five_year_plan_goal
				scope:journal_entry = {
				is_goal_complete = yes
			}
		}
		in_default = no
	}

    on_complete = { #如果完成
		add_modifier = {
			name = resource_empowerment #资源赋能
			years = 20
		}
	}

	current_value = {
		value = root.gdp
	}

	goal_add_value = {
		value = root.gdp
		multiply = 0.25
	}

	fail = {
	}

	progressbar = yes
}
