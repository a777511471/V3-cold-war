je_secularise = { #世俗化

	icon = "gfx/interface/icons/law_icons/无神论.dds"

	group = je_group_historical_content

    is_shown_when_inactive = { #显示条件
		NOT = {
			has_law = law_type:law_theocracy #神权制
		}			
        OR = {	
			has_law = law_type:law_state_religion #国教至上
			has_law = law_type:law_freedom_of_conscience #信仰自由			
		}	
    }
	possible = { #激活条件
		NOT = {
			has_law = law_type:law_theocracy #神权制
		}			
        OR = {	
			has_law = law_type:law_state_religion #国教至上
			has_law = law_type:law_freedom_of_conscience #信仰自由			
		}		
	}

	complete = {   
		has_technology_researched = tech_modernism #现代主义 
		NOT = {
			has_law = law_type:law_state_religion #国教至上
			has_law = law_type:law_freedom_of_conscience #信仰自由
			has_law = law_type:law_islamic_law_system #宗教法系
			has_law = law_type:law_religious_schools #教会学校
			has_law = law_type:law_charitable_health_system #慈善医院
			has_law = law_type:law_church_environmental_organizations #教会环保组织			
		}		                 				
	}

	on_complete = { #如果完成
		add_modifier = {
			name = secularise #世俗化
			years = 20          
		}
	}

	fail = { #失败条件		
		has_law = law_type:law_theocracy #神权制
	}    
}

		    has_law = law_type:law_council_republic #委员会共和制

            has_law = law_type:law_cooperative_ownership #合作社

je_collectivized_agriculture = { #集体化农业

	icon = "gfx/interface/icons/law_icons/collectivized_agriculture.dds"

	group = je_group_internal_affairs

    is_shown_when_inactive = { #显示条件

		NOT = { has_law = law_type:law_collectivized_agriculture } #没有集体化农业      
    }
	possible = { #激活条件
		has_law = law_type:law_council_republic #委员会共和制		
		NOT = { has_law = law_type:law_collectivized_agriculture } #没有集体化农业            
	}

	complete = {
		has_law = law_type:law_collectivized_agriculture #集体化农业	
		NOT = { has_law = law_type:law_land_based_taxation } #没有土地税制法律
		any_scope_state = {
			is_incorporated = yes
			any_scope_building = {
				OR = {
					is_building_type = building_rice_farm #水稻农场
					is_building_type = building_wheat_farm #小麦农场	 
					is_building_type = building_maize_farm #玉米农场
					is_building_type = building_rye_farm #黑麦农场	                       	 
				}
				level >= 20
			}
		}			
	}

	on_complete = {
		add_modifier = {
			name = land_reform
			years = 15
		}
	}

	fail = { #失败条件		
		NOT = { has_law = law_type:law_council_republic } #委员会共和制	
	}    
}
