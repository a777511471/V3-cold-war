je_algerian_affair_main = { #稳定阿尔及利亚局势
	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	group = je_group_historical_content

	complete = {
		NOT = {
        		has_law = law_type:law_colonial_exploitation
		}			
		every_scope_state = {
			limit = {
				OR = {
					state_region = {
						is_homeland = cu:maghrebi
					}
					state_region = {
						is_homeland = cu:berber
					}
				}
			}
			radical_fraction = {
				value < 0.1
			}
            turmoil < 0.1
		}
	}

	on_complete = { #如果完成
		trigger_event = {
			id = algerian_war.3
		}
	}
	fail = {
		OR = {
			any_scope_state = {
                OR = {
                    state_region = {
                        is_homeland = cu:maghrebi
                    }
                    state_region = {
                        is_homeland = cu:berber
                    }
                }
                OR = {
                    radical_fraction = {
                        value >= 0.40
                        culture = cu:maghrebi
                    }
                    radical_fraction = {
                        value >= 0.40
                        culture = cu:berber
                    }
                    turmoil >= 0.40
                }
			}
            highest_secession_progress > 0.5
		}
	}

	on_fail = {
		trigger_event = {
			id = algerian_war.2
		}
	}
	timeout = 2000

	invalid = {
		exists = c:ALD
	}

	on_timeout = {
		trigger_event = {
			id = algerian_war.2
		}
	}

	weight = 10000
	
	should_be_pinned_by_default = yes 
}
