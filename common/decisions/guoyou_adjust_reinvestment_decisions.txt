#投资池
adjust_reinvestment = {
	is_shown = {
		NOT = { has_modifier = adjust_reinvestment}
		NOT = { has_law = law_type:law_laissez_faire}
		NOT = { has_law = law_type:law_command_economy}
		NOT = { is_ai = no }		
		is_ai = no
	}

	when_taken = {
		add_modifier = {
			name = adjust_reinvestment
			months = -1
		}
	}
	
	ai_chance = {
        base = 0
    }	
}

stop_adjust_reinvestment = {
	is_shown = {
		has_modifier = adjust_reinvestment
	}

	when_taken = {
		remove_modifier = adjust_reinvestment
	}
	
	ai_chance = {
        base = 100
    }	
}

adjust_reinvestment_very_low = {
	is_shown = {
		has_modifier = adjust_reinvestment
		OR = {
			has_modifier = reinvestment_modifier_low
			has_modifier = reinvestment_modifier_medium
			has_modifier = reinvestment_modifier_high
			has_modifier = reinvestment_modifier_very_high
		}
	}

	when_taken = {
		remove_modifier = reinvestment_modifier_low
		remove_modifier = reinvestment_modifier_medium
		remove_modifier = reinvestment_modifier_high
		remove_modifier = reinvestment_modifier_very_high
	}
	
    ai_chance = {
        base = 0
    }	
}
